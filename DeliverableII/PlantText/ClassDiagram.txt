' ======================= Domain Classes =======================
class User {
  +userId: String
  +email: String
  +displayName: String
}
class AITravelAgent {
  +planTrip(request:TripRequest):TripPlanSummary
  +generateItinerary(request: Trip,prefs: Preference): Itinerary
}

class Preference {
  +userId: String
  +descriptions: String
}
class TripRequest {
  +origin: String
  +destination: String
  +startDate: Date
  +endDate: Date
  +budget: Float
  +preferences: Preference
  +transportMode: String  <<flight|driving|both>>
  +homeCurrency: String
  +destinationCurrency: String

}
class Trip {
  +tripId: String
  +userId: String
  +origin: String
  +destination: String
  +startDate: Date
  +endDate: Date
  +budget: Float
  +homeCurrency: String
  +destinationCurrency: String
  +createdAt: DateTime
  +status: String
  +planSummary: String
}

class Itinerary {
  +itineraryId: String
  +tripId: String
  +summary: String
  +generatedAt: DateTime
}

class ItineraryDay {
  +itineraryId: String
  +dayNumber: Integer
  +date: Date
  +weatherSummary: String
  +activities: List<String>
  +notes: String
}

class Location {
  +latitude: Float
  +longitude: Float
  +cityName: String
  +country: String
}

class WeatherForecast {
  +tripId: String
  +date: Date
  +tempHigh: Float
  +tempLow: Float
  +condition: String
  +precipChance: Float
}

class FlightOption {
  +tripId: String
  +carrier: String
  +flightNumber: String
  +departureTime: DateTime
  +arrivalTime: DateTime
  +price: Float
}

class RouteEstimate {
  +tripId: String
  +mode: String
  +durationMinutes: Integer
  +distanceKm: Float
}

class Attraction {
  +tripId: String
  +name: String
  +type: String
  +rating: Float
  +address: String
}

class CurrencyRate {
  +tripId: String
  +baseCurrency: String
  +targetCurrency: String
  +rate: Float
  +retrievedAt: DateTime
}


class AuthService {
  +generateAuthUrl(state: String): String
  +handleCallback(code: String, state: String): User
  +logout(userId: String): void
}


class WeatherService {
  +getForecast(dest: Location, dates: List<Date>): List<WeatherForecast>
}

class FlightService {
  +getFlights(origin: String, dest: String, dates: List<Date>): List<FlightOption>
}

class PlacesService {
  +geocode(destination: String): Location
  +getPointsOfInterest(loc: placeID): List<Attraction>
  +estimateRoute(origin: String, dest: String): RouteEstimate
}

class CurrencyService {
  +getRate(homeCurrency: String, destCurrency: String): CurrencyRate
}



class TripRepository {
  +saveTrip(trip: Trip, prefs: Preference, itinerary: Itinerary, days: List<ItineraryDay>): void
  +findTripsByUser(userId: String): List<Trip>
  +findTripWithDetails(tripId: String, userId: String): Trip
  +deleteTrip(userId: String, tripId: String): void
}

class PreferenceRepository {
  +getPreference(userId: String): Preference
  +savePreference(pref: Preference): void
}

class UserRepository {
  +upsertUser(profile: User): User
  +findUser(userId: String): User
}

class OpenAIService {
  +generateRecommendations(message:String):TripPlanSummary
}
